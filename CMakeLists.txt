    list(APPEND COMMON_INCLUDES "${MM_DIR}/${_path}")

function(blinky_game_library name)
    add_library(${name} STATIC ${ARGN})
    target_include_directories(${name} PRIVATE ${COMMON_INCLUDES})
    target_compile_options(${name} PRIVATE -Os -w -fpermissive)
endfunction()

        z_camera z_lights z_play z_room z_scene z_view)
    if(EXISTS "${MM_DIR}/src/code/${_hot}.c")
        set_source_files_properties("${MM_DIR}/src/code/${_hot}.c" PROPERTIES COMPILE_OPTIONS -O2)
    endif()
endforeach()

        ship/resource/ResourceManager.h ship/window/Window.h spdlog/spdlog.h

    target_precompile_headers(mm_enhancements PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:<${_header}$<ANGLE-R>>")

    "$<$<COMPILE_LANGUAGE:CXX>:${MM_DIR}/2s2h/GameInteractor/GameInteractor.h>")

    else()

    endif()

        target_precompile_headers(mm_enhancements PRIVATE "$<$<COMPILE_LANGUAGE:${_language}>:${MM_DIR}/include/${_header}.h>")
    endforeach()
